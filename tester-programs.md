type Order = {
    id: Int,
    item: String,
    quantity: Int,                          // [R1-F1: ambiguity — trailing comma in record types]

}

error InvalidOrder = { reason: String }     // [R1-F2: contradiction — = vs no = in error decl]

// --- functions ---

fn describe_status(s: :pending | :processing | :complete | :cancelled) -> String {
    match s {
        :pending    => "Waiting to be processed"
        :processing => "Currently being processed"
        :complete   => "Finished"
        :cancelled  => "Cancelled"
    }
}

fn validate_order(order: Order) -> Order {
    if order.quantity <= 0 {
        raise(InvalidOrder { reason: "quantity must be positive" })  // [R1-F3: contradiction — raise() vs raise]
    }
    order

        valid => { ...valid, status: :processing }
        InvalidOrder { reason } => {
            IO.println("Rejected: {reason}")
            { ...order, status: :cancelled }   // [R1-F4: spec gap — IO return type / unit]
        }
    }
}

    let o = Order {                          // [R1-F5: ambiguity — Point{} vs { } with type annotation]
        id: 1,
        item: "Widget",
        quantity: 3,
        status: :pending,
    }
    let result = process_order(o)
    IO.println("Order {result.id}: {describe_status(result.status)}")
}

**Narration:** The spec shows UFCS where `self` is the first parameter name. Generics are covered

type Score = { player: String, value: Int }

// A generic filter-then-transform pipeline
fn above_threshold<T>(self: List<T>, threshold: Int, get_value: fn(T) -> Int) -> List<T> {
    self |> filter { item => get_value(item) > threshold }
}                                           // [R1-F6: spec gap — constraint syntax for generic fns]

fn summarise(scores: List<Score>) -> String {
    let winners = scores |> above_threshold(50, fn(s) => s.value)  // [R1-F7: ambiguity — lambda syntax]

### Program 2c — Concurrency + State + Effects

**Domain:** A concurrent counter: spawn workers that update shared state, collect results.

**Narration:** Concurrency from 07-concurrency.md. `Concurrency.scope`, `State<T>`, `spawn`.
The spec is detailed here. Let me write a realistic example.

```bounce
// --- types ---

type WorkResult = { worker_id: Int, processed: Int }

// --- worker ---

fn process_chunk(worker_id: Int, chunk: List<Int>, counter: State<Int>) -> WorkResult {
    mut processed = 0                        // [R2-F18: ambiguity — mut in regular (non-generator) fn?]
    chunk |> for_each { item =>
        counter.update { n => n + item }
        processed = processed + 1            // [R2-F19: ambiguity — assignment to mut var in closure?]
    }
    WorkResult { worker_id, processed }
}

// --- main ---

fn run_concurrent(items: List<Int>) -> Int {
    let chunk_size = 100
    let chunks = items |> chunk(chunk_size)  // [R2-F20: spec gap — chunk() function not defined]

    Concurrency.scope { s =>
        let total = s.state(0)              // State<Int> initial value 0

        let tasks = chunks
            |> map_with_index { chunk, i =>  // [R2-F21: spec gap — map_with_index not defined]
                s.spawn { process_chunk(i, chunk, total) }
            }

        tasks |> for_each { t => t.await() }  // [R2-F22: spec gap — Task<T>.await() syntax]

        total.read { n => n }
    }
}

test fn concurrent_counter_correct() {
    let items = range(0, 1000) |> collect   // [R2-F23: ambiguity — range returns Sequence, collect returns List]
    let result = run_concurrent(items)
    assert_eq(result, 499500)               // sum 0..999
}
```

// ====================================================================
// models/record.bounce
// ====================================================================

type RawRow = { line: Int, data: Map<String, String> }

type ProcessedRow = {
    id: Int
    name: String
    score: Float
    label: String
}

error ProcessingError =
    | MissingField  { row: Int, field: String }
    | InvalidValue  { row: Int, field: String, value: String }
    | WriteError    { row: Int, message: String }

// ====================================================================
// pipeline.bounce
// ====================================================================

fn parse_row(raw: RawRow) -> ProcessedRow {
    let id    = try Int.parse(raw.data["id"]) {         // [R3-F17: spec gap — Map index syntax]
        n => n
        :false => raise(MissingField { row: raw.line, field: "id" })  // [R3-F18: contradiction — :false for None]
    }
    let name  = raw.data["name"]                        // [R3-F19: friction — Map access returns Option, must handle]
    let score = try Float.parse(raw.data["score"]) {
        n => n
        :false => raise(InvalidValue { row: raw.line, field: "score", value: raw.data["score"] })
    }
    ProcessedRow {
        id,
        name: name |> unwrap_or("Unknown"),             // [R3-F20: spec gap — unwrap_or not defined]
        score,
        label: if score >= 80.0 { "pass" } else { "fail" },
    }
}

fn write_row(row: ProcessedRow) -> () {
    FileSystem.append(
        "output.csv",
        "{row.id},{row.name},{row.score},{row.label}\n",
    )                                                    // [R3-F21: spec gap — FileSystem.append not defined]
}

// ====================================================================
// app.bounce
// ====================================================================

fn process_jobs(input_path: String) -> () {
    let started_at = Time.now()

    let rows: Sequence<RawRow> & :finite = read_csv(input_path)  // [R3-F22: spec gap — read_csv not stdlib]
    let results = rows
        |> map { row =>
            try parse_row(row) {
                record => :ok { value: record }
                ProcessingError { ... } => :err { error: ProcessingError }  // [R3-F17: same catch-all issue]
            }
        }

    mut successes = 0
    mut failures  = 0

    results |> for_each { r =>
        match r {
            :ok { value } => {
                write_row(value)
                successes = successes + 1
            }
            :err { error } => {
                IO.println("Row failed: {error}")
                failures = failures + 1
            }
        }
    }

    let elapsed = Time.now() - started_at  // [R3-F23: spec gap — Instant arithmetic, result type is Duration?]
    IO.println("Done: {successes} ok, {failures} failed, {elapsed} elapsed")
}
