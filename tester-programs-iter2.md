### Program 1a — Decimal Arithmetic and Money Handling

**Domain:** Invoice line-item pricing with tax, discount, and penny-safe splitting.

**Narration:** The spec (01-primitives.md §2) defines Decimal literals with the `d` suffix and
mentions `Decimal.divide_distribute` for splitting and `Decimal.round(scale, mode)` for rounding.
I want to model a realistic invoice calculation: subtotals, percentage-based tax, discounts, and
splitting a bill evenly. This will exercise `Decimal` math, scale tracking, and the distribute API.

```bounce
// --- types ---

type LineItem = {
    name: String,
    unit_price: Decimal,
    quantity: Int,
}

type Invoice = {
    items: List<LineItem>,
    tax_rate: Decimal,                         // e.g. 0.08d for 8%
    discount_pct: Decimal,                     // e.g. 0.10d for 10%
}

// --- functions ---

fn line_total(item: LineItem) -> Decimal {
    item.unit_price * item.quantity             // [R4-F1: ambiguity — Decimal * Int: is Int auto-promoted?]
}

fn subtotal(invoice: Invoice) -> Decimal {
    invoice.items
        |> map { item => line_total(item) }
        |> fold(0.00d, { acc, v => acc + v })  // [R4-F2: guess — fold on a List vs Sequence]
}

fn apply_discount(amount: Decimal, pct: Decimal) -> Decimal {
    let discount = amount * pct
    let discounted = amount - discount
    discounted |> Decimal.round(2, :half_up)   // [R4-F3: ambiguity — Decimal.round is UFCS companion or module fn?]
}

fn apply_tax(amount: Decimal, rate: Decimal) -> Decimal {
    let tax = amount * rate
    (amount + tax) |> Decimal.round(2, :half_up)
}

fn split_bill(total: Decimal, ways: Int) -> List<Decimal> {
    Decimal.divide_distribute(total, ways)     // [R4-F4: spec gap — does divide_distribute accept a scale/rounding mode?]
}

fn main() {
    let inv = Invoice {
        items: [
            LineItem { name: "Widget", unit_price: 19.99d, quantity: 3 },
            LineItem { name: "Gadget", unit_price: 49.95d, quantity: 1 },
        ],
        tax_rate: 0.0875d,
        discount_pct: 0.10d,
    }

    let sub = subtotal(inv)                    // 109.92d  (19.99*3 + 49.95)
    let after_discount = apply_discount(sub, inv.discount_pct)
    let final_total = apply_tax(after_discount, inv.tax_rate)

    Terminal.println("Subtotal:  {sub}")        // [R4-F5: spec gap — Decimal display format in interpolation]
    Terminal.println("After discount: {after_discount}")
    Terminal.println("Final (with tax): {final_total}")

    let splits = split_bill(final_total, 3)
    splits |> each { s =>                      // [R4-F6: guess — `each` on List directly?]
        Terminal.println("  Share: {s}")
    }
}
```

### Program 1b — Time Handling

**Domain:** Meeting scheduler that checks timezone conversions, computes durations, and detects overlaps.

**Narration:** The spec (01-primitives.md §2) distinguishes `Datetime` (calendar, timezone-aware),
`Instant` (monotonic), `Duration` (span), and `Date` (calendar date). I'll exercise timezone conversion,
duration arithmetic, and the `Time` effect for getting current time. The spec says
`Datetime` math respects DST, and `Instant` only supports fixed `Duration` addition.

```bounce
// --- types ---

type Meeting = {
    title: String,
    start: Datetime,
    duration: Duration,
}

// --- functions ---

fn end_time(m: Meeting) -> Datetime {
    m.start + m.duration                       // [R4-F7: ambiguity — Datetime + Duration: does this respect DST?]
}

fn overlaps(a: Meeting, b: Meeting) -> Bool {
    let a_end = end_time(a)
    let b_end = end_time(b)
    a.start < b_end && b.start < a_end        // [R4-F8: ambiguity — comparison operators on Datetime]
}

fn to_timezone(dt: Datetime, tz: String) -> Datetime {
    dt.in_timezone(tz)                         // [R4-F9: spec gap — no API for timezone conversion on Datetime]
}

fn minutes_until(meeting: Meeting) -> Duration {
    let now = Time.now_utc()                   // [R4-F10: ambiguity — Time.now_utc() returns Datetime per spec]
    meeting.start - now                        // [R4-F11: spec gap — Datetime - Datetime -> Duration not specified]
}

fn format_duration(d: Duration) -> String {
    let total_minutes = d.as_minutes()         // [R4-F12: spec gap — Duration accessor methods not specified]
    let hours = total_minutes / 60
    let mins = total_minutes % 60
    "{hours}h {mins}m"
}

fn main() {
    let standup = Meeting {
        title: "Daily Standup",
        start: @2026-03-13T09:00:00[America/New_York],
        duration: 15m,
    }

    let review = Meeting {
        title: "Code Review",
        start: @2026-03-13T09:10:00[America/New_York],
        duration: 45m,
    }

    let london_time = to_timezone(standup.start, "Europe/London")
    Terminal.println("Standup in London: {london_time}")     // [R4-F13: spec gap — Datetime display format]

    if overlaps(standup, review) {
        Terminal.println("WARNING: {standup.title} and {review.title} overlap!")
    }

    let until = minutes_until(standup)
    Terminal.println("Time until standup: {format_duration(until)}")
}
```

**Domain:** A small "accounts" library package with opaque types, visibility tiers, and companion functions.

**Narration:** The spec (methods-and-packages.md, modules-and-imports.md) defines `package.bounce`

```bounce
// ============================
// FILE: accounts/package.bounce
// ============================

package {
    name: accounts
    version: "1.0.0"
    license: MIT

    deps {
        std: "1.2"
    }

    lints {
        std/unused: :error
    }
}

export opaque type Account with { create, display, deposit, withdraw, balance, id }
export nominal type AccountId with { display }          // [R4-F23: ambiguity — can nominal types have companions?]
export type TransferError                               // structural, fully transparent
export fn transfer

// ============================
// FILE: accounts/account.bounce
// ============================

nominal type AccountId = Int                            // [R4-F24: guess — nominal type wrapping Int, construct with AccountId(42)]

opaque type Account = {
    id: AccountId,
    holder: String,
    balance_cents: Int,
    internal created_at: Datetime,                      // [R4-F25: ambiguity — can opaque type fields have `internal` visibility?]
}

error InsufficientFunds = { account_id: AccountId, requested: Int, available: Int }
error InvalidAmount = { amount: Int, reason: String }

type TransferError =
    | InsufficientFunds { account_id: AccountId, requested: Int, available: Int }
    | InvalidAmount { amount: Int, reason: String }

// --- companion functions ---

pub fn create(holder: String, initial_deposit: Int) -> Account {
    if initial_deposit < 0 {
        raise(InvalidAmount { amount: initial_deposit, reason: "initial deposit cannot be negative" })
    }
    let id = AccountId(Random.int(1, 1000000))         // [R4-F26: ambiguity — AccountId construction from Int]
    Account {
        id: id,
        holder: holder,
        balance_cents: initial_deposit,
        created_at: Time.now_utc(),
    }
}

pub fn display(self: Account) -> String {
    "Account({self.id |> display}) - {self.holder}: {self.balance_cents} cents"
}                                                       // [R4-F27: guess — calling display on AccountId inside Account's display]

pub fn deposit(self: Account, amount: Int) -> Account {
    if amount <= 0 {
        raise(InvalidAmount { amount: amount, reason: "deposit must be positive" })
    }
    { ...self, balance_cents: self.balance_cents + amount }
}

pub fn withdraw(self: Account, amount: Int) -> Account {
    if amount <= 0 {
        raise(InvalidAmount { amount: amount, reason: "withdrawal must be positive" })
    }
    if amount > self.balance_cents {
        raise(InsufficientFunds {
            account_id: self.id,
            requested: amount,
            available: self.balance_cents,
        })
    }
    { ...self, balance_cents: self.balance_cents - amount }
}

pub fn balance(self: Account) -> Int {
    self.balance_cents
}

pub fn id(self: Account) -> AccountId {
    self.id
}

// --- companion for AccountId ---

pub fn display(self: AccountId) -> String {
    "ACCT-{self}"                                       // [R4-F28: spec gap — how to access inner value of nominal type in interpolation]
}

// --- standalone function ---

pub fn transfer(from: Account, to: Account, amount: Int) -> (Account, Account) {
    let debited = from |> withdraw(amount)              // [R4-F29: guess — UFCS pipeline with companion fn]
    let credited = to |> deposit(amount)
    (debited, credited)
}

// ============================
// FILE: accounts/account.test.bounce
// ============================

}

test fn withdraw_insufficient_funds_raises() {
    let acct = Account.create("Bob", 1000)
    assert_raises<InsufficientFunds> {
        acct |> Account.withdraw(2000)                  // [R4-F30: friction — do I say Account.withdraw or just withdraw?]
    }
}

test fn transfer_moves_funds() {
    let alice = Account.create("Alice", 10000)
    let bob = Account.create("Bob", 0)
    let (alice2, bob2) = transfer(alice, bob, 3000)
    assert_eq(alice2.balance(), 7000)
    assert_eq(bob2.balance(), 3000)
}

// ============================
// FILE: consumer/app.bounce
// ============================

fn main() {
    let a = Account.create("Alice", 100_00)             // [R4-F31: ambiguity — underscore separators in integer literals?]
    let b = Account.create("Bob", 50_00)

    try {
        let (a2, b2) = transfer(a, b, 75_00)

        Terminal.println("Alice: {a2 |> Account.display()}")
        Terminal.println("Bob: {b2 |> Account.display()}")
    } catch {                                           // [R4-F32: friction — try/catch vs try { expr } { arms }]
```

### Program 2c — Sequence Operators (map, filter, scan, debounce, merge, zip)

**Domain:** Real-time sensor data processing pipeline with temperature readings.

**Narration:** The spec (07-concurrency.md) defines `Sequence<T>` as the universal abstraction with
finiteness tags, pipeline operators (map, filter, scan, debounce, merge, zip), and terminal operations
(collect, each, fold). I'll build a sensor monitoring pipeline.

```bounce
// --- types ---

type SensorReading = {
    sensor_id: String,
    temperature: Float,
    timestamp: Instant,
}

type Alert =
    | :high_temp { sensor_id: String, temp: Float }
    | :rapid_change { sensor_id: String, delta: Float }
    | :sensor_offline { sensor_id: String }

type SensorStats = {
    sensor_id: String,
    avg_temp: Float,
    min_temp: Float,
    max_temp: Float,
    reading_count: Int,
}

// --- generator ---

fn simulated_sensor(id: String, base_temp: Float) -> Sequence<SensorReading> & :infinite {
    mut temp = base_temp
    loop {
        let jitter = Random.float() * 2.0 - 1.0       // random [-1.0, 1.0)
        temp = temp + jitter
        yield SensorReading {
            sensor_id: id,
            temperature: temp,
            timestamp: Time.now(),
        }
        Time.sleep(100ms)
    }
}

// --- processing pipeline ---

fn detect_alerts(readings: Sequence<SensorReading> & :infinite) -> Sequence<Alert> & :infinite {
    let high_temps = readings
        |> filter { r => r.temperature > 100.0 }
        |> map { r => :high_temp { sensor_id: r.sensor_id, temp: r.temperature } }

    let rapid_changes = readings
        |> scan(                                       // [R4-F45: ambiguity — scan signature: scan(init, fn) or scan(fn)?]
            { last_temp: 0.0, delta: 0.0 },
            { state, reading =>
                let delta = (reading.temperature - state.last_temp) |> Float.abs()  // [R4-F46: spec gap — Float.abs() not specified]
                { last_temp: reading.temperature, delta: delta }
            },
        )
        |> filter { state => state.delta > 5.0 }
        |> map { state => :rapid_change { sensor_id: "unknown", delta: state.delta } }  // [R4-F47: friction — lost sensor_id in scan state]

    merge([high_temps, rapid_changes])                 // [R4-F48: ambiguity — merge requires same type: Alert here]
}

fn rolling_average(
    readings: Sequence<SensorReading> & :infinite,
    window: Duration,
) -> Sequence<Float> & :infinite {
    readings
        |> buffer(window)                              // groups readings into time windows
        |> map { batch =>
            let sum = batch |> map { r => r.temperature } |> fold(0.0, { a, v => a + v })
            sum / batch.length() |> Int.to_float()     // [R4-F49: spec gap — List.length() returns Int, need to convert for Float division]
        }
}

fn monitor_sensors() {
    Concurrency.scope { s =>
        // Create sensor streams
        let sensor_a = simulated_sensor("A", 70.0)
        let sensor_b = simulated_sensor("B", 85.0)

        // Merge all sensor streams
        let all_readings = merge([sensor_a, sensor_b])

        // Set up alert detection
        let alerts = detect_alerts(all_readings)       // [R4-F50: friction — all_readings consumed by merge, can it be shared?]

        // Rolling average per sensor
        let avg_a = sensor_a                           // [R4-F51: spec gap — reusing a Sequence: is it replayed or shared?]
            |> filter { r => r.sensor_id == "A" }
            |> rolling_average(5s)

        // Zip alerts with timestamps
        let timed_alerts = zip(alerts, Time.every(1s)) // [R4-F52: ambiguity — zip pairs 1:1; does it buffer or drop?]
            |> map { (alert, tick) => { alert: alert, at: tick } }

        // Debounce alerts (don't spam)
        let debounced = timed_alerts
            |> debounce(10s)                           // [R4-F53: ambiguity — debounce emits after quiet period; how does this work with tagged unions?]

        // Terminal: consume alerts for 1 minute
        let stop = Time.after(1m)

        s.spawn {
            debounced
                |> take_until(stop)                    // [R4-F54: guess — take_until converts infinite to finite]
                |> each { event =>
                    match event.alert {
                        :high_temp { sensor_id, temp } =>
                            Log.warn("HIGH TEMP on {sensor_id}: {temp}°F")
                        :rapid_change { sensor_id, delta } =>
                            Log.warn("RAPID CHANGE on {sensor_id}: Δ{delta}°F")
                        :sensor_offline { sensor_id } =>
                            Log.error("OFFLINE: {sensor_id}")
                    }
                }
        }

        // Collect stats from rolling average (finite window)
        s.spawn {
            avg_a
                |> take(60)                            // first 60 readings
                |> collect                             // [R4-F55: ambiguity — collect is terminal: collect() or collect?]
                |> { averages =>
                    let final_avg = averages |> fold(0.0, { a, v => a + v }) / averages.length() |> Int.to_float()
                    Log.info("Sensor A average: {final_avg}°F")
                }
        }

        // Wait for all spawned tasks
    }
}
```

```bounce
// --- valid transitions table ---

pure fn is_valid_transition(state: OrderState, event: OrderEvent) -> Bool {
    match (state, event) {                             // [R4-F58: ambiguity — tuple matching: match on (A, B)?]
        (:draft, :submitted { ... })            => true
        (:submitted { ... }, :payment_initiated { ... }) => true
        (:payment_pending { ... }, :payment_confirmed { ... }) => true
        (:paid { ... }, :picking_started { ... }) => true
        (:picking { ... }, :packing_complete { ... }) => true
        (:packed { ... }, :shipped { ... })     => true
        (:shipped { ... }, :delivered { ... })  => true
        // Cancellation from early states
        (:draft, :cancelled { ... })            => true
        (:submitted { ... }, :cancelled { ... }) => true
        (:payment_pending { ... }, :cancelled { ... }) => true
        // Refund from paid/shipped/delivered
        (:paid { ... }, :refund_initiated { ... }) => true
        (:shipped { ... }, :refund_initiated { ... }) => true
        (:delivered { ... }, :refund_initiated { ... }) => true
        _ => false                                     // [R4-F59: guess — wildcard catch-all in tuple match]
    }
}

// --- state transition logic ---

fn apply_event(order: Order, event: OrderEvent) -> Order {
    if !is_valid_transition(order.state, event) {
        raise(TransitionError {
            order_id: order.id,
            from_state: describe_state(order.state),
            to_event: describe_event(event),
            reason: "invalid transition",
        })
    }

    let now = Time.now_utc()
    let new_state = match event {
        :submitted { at }           => :submitted { submitted_at: at }
        :payment_initiated { payment_id, at } => :payment_pending { payment_id: payment_id }
        :payment_confirmed { at }   => {
            match order.state {
                :payment_pending { payment_id } =>
                    :paid { paid_at: at, payment_id: payment_id }
                _ => panic("unreachable: validated above")    // [R4-F60: ambiguity — panic syntax: panic("msg") or panic "msg"?]
            }
        }
        :picking_started { warehouse, picker_id, at } =>
            :picking { warehouse: warehouse, picker_id: picker_id }
        :packing_complete { weight_kg, at } =>
            :packed { packed_at: at, weight_kg: weight_kg }
        :shipped { carrier, tracking, at } =>
            :shipped { carrier: carrier, tracking: tracking, shipped_at: at }
        :delivered { at }           => :delivered { delivered_at: at }
        :cancelled { reason, at }   => :cancelled { reason: reason, cancelled_at: at }
        :refund_initiated { refund_id, at } => :refunded { refund_id: refund_id, refunded_at: at }
        :created { ... }            => order.state     // no-op, already created
    }

    {
        ...order,
        state: new_state,
        updated_at: now,
        history: order.history ++ [event],             // [R4-F61: spec gap — List concatenation: ++ or List.append?]
    }
}

pub fn display(self: Order) -> String {
    "Order {self.id} [{describe_state(self.state)}] - {self.items |> List.length()} items"
}                                                      // [R4-F62: ambiguity — List.length() vs .len() vs |> length]

pub fn state(self: Order) -> OrderState { self.state }
pub fn id(self: Order) -> String { self.id }
pub fn items(self: Order) -> List<OrderItem> { self.items }

pub fn total(self: Order) -> Int {
    self.items
        |> map { item => item.quantity * item.unit_price_cents }
        |> fold(0, { acc, v => acc + v })
}

pub fn transition(self: Order, event: OrderEvent) -> Order {

}

fn notify_customer(order: Order, event: OrderEvent) {
    match event {
        :shipped { carrier, tracking, ... } => {
            Http.post_json(
                "https://notifications.internal/send",
                Notification.email({                   // [R4-F63: guess — constructing a view inline from anonymous record]

            )
        }

            Http.post_json(
                "https://notifications.internal/send",
                Notification.email({
                    to: order.customer_id,
                    subject: "Your order has been delivered",
                    body: "Order {order.id} was delivered.",
                }),
            )
        }

fn persist_event(order: Order, event: OrderEvent) {
    Database.execute(
        "INSERT INTO order_events (order_id, event_type, payload, created_at) VALUES ($1, $2, $3, $4)",
        [
            :text { v: order.id },
            :text { v: describe_event(event) },
            :text { v: event |> to_json() },           // [R4-F65: spec gap — to_json on union variant: is this auto-derived?]
            :text { v: Time.now_utc() |> Datetime.to_iso8601() },
        ],
    )
}

// --- public entry point ---

pub fn process_event(order: Order, event: OrderEvent) -> Order {
    Log.info_fields("Processing event", {
        "order_id": order.id,
        "event": describe_event(event),
        "current_state": describe_state(order.state),
    })

    let updated = order |> transition(event)

    persist_event(updated, event)
    notify_customer(updated, event)

    Log.info_fields("Transition complete", {
        "order_id": updated.id,
        "new_state": describe_state(updated.state),
    })

    updated
}
```

```bounce
pub fn start() -> ChatServer {

        let rooms = s.registry<RoomId, RoomState>(
            init: { room_id =>
                RoomState {
                    members: Set.new(),
                    history: [],
                    created_at: Time.now_utc(),
                }
            },
        )

            capacity: 1000,
            overflow: :drop_oldest,
        )

        s.detach {
            for tick in Time.every(30s) {
                let now = Time.now()
                sessions.update { sess_map =>
                    sess_map |> Map.filter { username, session =>  // [R4-F77: spec gap — Map.filter: does it take (K, V) -> Bool?]

                        idle < 300                     // 5 minutes
                    }
                }
            }
        }

        // Background task: persist message history
        s.detach {
            let sub = message_bus.subscribe()
            for event in sub {
                match event {
                    :message { msg } => {
                        Database.execute(
                            "INSERT INTO messages (room, sender, content, sent_at) VALUES ($1, $2, $3, $4)",

import { ChatServer, ChatMessage, ChatError } from chat

pub fn handle_connection(server: ChatServer, ws: WsConnection, username: String) {
    let rate_limiter = Concurrency.scope { s =>         // [R4-F82: friction — nested scope for per-connection state]
        let rl = s.state(RateLimiter.new(RateLimiterConfig {
            max_messages: 20,
            window: 60s,
        }))

        let events_sub = server.message_bus.subscribe()

        // Outbound: forward relevant server events to this client's websocket
        let outbound = s.spawn {
            for event in events_sub {
                let should_send = match event {
                    :message { msg } => {
                        server.sessions.read { sess =>
                            match sess |> Map.get(username) {
                                :false => false
                                session => session.rooms |> Set.contains(msg.room)  // [R4-F83: ambiguity — Option narrowing: session is (UserSession & :true) here]
                            }
                        }
                    }
                    :user_joined { room, ... } => is_in_room(server, username, room)
                    :user_left { room, ... }   => is_in_room(server, username, room)
                    :room_list { ... }         => true
                    :error { ... }             => true
                    :pong                      => true
                }

                if should_send {
                    ws.send(event |> to_json())         // [R4-F84: spec gap — to_json on union: needs a view?]
                }
            }
        }

        // Inbound: process client commands from websocket
        let inbound = s.spawn {
            for ws_msg in ws.messages() {
                match ws_msg {
                    :text { data } => {
                        let cmd = data |> Json.parse<ClientCommand>  // [R4-F85: spec gap — Json.parse on a union type: needs input declaration?]
                        try handle_command(server, username, cmd, rl) {
                            _ => ()
                            RateLimitExceeded { ... } => {
                                ws.send((:error { code: 429, message: "rate limited" }) |> to_json())
                            }
                            ChatError { ... } => {     // [R4-F86: ambiguity — catching grouped error by group name]
                                ws.send((:error { code: 400, message: "error" }) |> to_json())
                            }
                        }
                    }
                    :binary { ... } => ()              // ignore binary
                    :close { code, reason } => {
                        Log.info("Client {username} disconnected: {reason}")
                        s.cancel()                     // end connection scope
                    }
                }
            }
        }

        // Heartbeat: send ping every 30s
        s.detach {
            for _ in Time.every(30s) {                 // [R4-F87: ambiguity — ignoring yielded value with _]
                ws.send((:pong) |> to_json())          // [R4-F88: friction — bare atom in parens for to_json: (:pong) vs :pong]
            }
        }

        // Wait for either inbound or outbound to finish
        for result in select(inb: inbound, outb: outbound) {
            match result {
                :inb { ... } => s.cancel()
                :outb { ... } => s.cancel()
            }
        }
    }

    // Cleanup: remove from all rooms
    cleanup_user(server, username)
}

fn handle_command(
    server: ChatServer,
    username: String,
    cmd: ClientCommand,
    rl: State<RateLimiter>,
) {
    // Update last activity
    server.sessions.update { sess =>
        match sess |> Map.get(username) {
            :false => sess
            session => sess |> Map.set(username, { ...session, last_activity: Time.now_utc() })  // [R4-F89: spec gap — Map.set / Map.insert API not specified]
        }
    }

    match cmd {
        :join { room } => {
            let room_state = server.rooms.get(room)    // creates if not exists
            room_state.update { r =>
                { ...r, members: r.members |> Set.insert(username) }  // [R4-F90: spec gap — Set.insert API not specified]
            }
            server.sessions.update { sess =>
                match sess |> Map.get(username) {
                    :false => sess
                    session => sess |> Map.set(username, {
                        ...session,
                        rooms: session.rooms |> Set.insert(room),
                    })
                }
            }
            server.message_bus.publish(:user_joined { room: room, user: username })
        }

        :leave { room } => {
            let room_state = server.rooms.get(room)
            room_state.update { r =>
                { ...r, members: r.members |> Set.remove(username) }
            }
            server.message_bus.publish(:user_left { room: room, user: username })
        }

        :send_message { room, content } => {
            // Rate limiting
            rl.update { limiter =>
                limiter |> RateLimiter.check(Time.now())
            }

            let msg = ChatMessage {
                room: room,
                sender: username,
                content: content,
                sent_at: Time.now_utc(),
            }

            // Add to room history
            let room_state = server.rooms.get(room)
            room_state.update { r =>
                { ...r, history: r.history ++ [msg] |> List.take_last(100) }  // [R4-F91: spec gap — List.take_last not specified]
            }

            server.message_bus.publish(:message { msg: msg })
        }

        :list_rooms => {
            let room_ids = server.rooms.keys()         // Registry.keys() -> List<K>
            server.message_bus.publish(:room_list { rooms: room_ids })  // [R4-F92: friction — publishing to all subscribers when only one client asked]
        }

        :ping => {
            // Handled by heartbeat, but acknowledge
        }
    }
}

fn is_in_room(server: ChatServer, username: String, room: RoomId) -> Bool {
    server.sessions.read { sess =>
        match sess |> Map.get(username) {
            :false => false
            session => session.rooms |> Set.contains(room)
        }
    }
}

fn cleanup_user(server: ChatServer, username: String) {
    server.sessions.read { sess =>
        match sess |> Map.get(username) {
            :false => ()
            session => {
                session.rooms |> each { room =>
                    let room_state = server.rooms.get(room)
                    room_state.update { r =>
                        { ...r, members: r.members |> Set.remove(username) }
                    }
                    server.message_bus.publish(:user_left { room: room, user: username })
                }
            }
        }
    }                                                  // [R4-F93: friction — read closure has side effects (rooms.get, publish) — but read requires pure fn!]

    server.sessions.update { sess =>
        sess |> Map.remove(username)
    }
}
```
